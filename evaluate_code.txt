import numpy as np, rasterio
from rasterio.warp import reproject, Resampling

def read1(p):
    with rasterio.open(p) as ds:
        a = ds.read(1).astype(np.float32)
        prof = ds.profile
    return a, prof

def to_same_grid(src_arr, src_prof, ref_prof):
    dst = np.empty((ref_prof["height"], ref_prof["width"]), np.float32)
    reproject(
        source=src_arr,
        destination=dst,
        src_transform=src_prof["transform"],
        src_crs=src_prof["crs"],
        dst_transform=ref_prof["transform"],
        dst_crs=ref_prof["crs"],
        resampling=Resampling.bilinear,
    )
    return dst

def down2x_mean(x):
    h,w=x.shape
    h2=(h//2)*2; w2=(w//2)*2
    x=x[:h2,:w2]
    return x.reshape(h2//2,2,w2//2,2).mean(axis=(1,3))

# 入力5mから作った4m（推論時のlr4）を保存してない場合は、
# ここでは「元の5m」を4mへリサンプルしたものを“入力4m相当”として使う
p5="/home/s1300211/Documents/datasets/test/center_tile.tif"
psr="/home/s1300211/Documents/EBCF-CDEM/moon_data/out_center_tile_2m_v5.tif"  # ←あなたのSR出力に合わせて変更

x5, prof5 = read1(p5)
sr2, prof2 = read1(psr)

# 5m→4mを作る（評価用）
with rasterio.open(p5) as ds:
    prof4 = ds.profile.copy()
# 4mグリッドを作るために、sr2を4mに落とした後に合わせる（簡便）
sr4 = down2x_mean(sr2)

# sr4 を 5m由来の4m（リサンプル）と同一グリッドに揃える
# まず 5m→4m を作る
x4 = np.empty((int(round(prof5["height"]*5/4)), int(round(prof5["width"]*5/4))), np.float32)
# 4m用profileを作る
prof4 = prof5.copy()
from rasterio.transform import Affine
t5=prof5["transform"]
prof4.update(height=x4.shape[0], width=x4.shape[1],
             transform=Affine(4.0, t5.b, t5.c, t5.d, -4.0, t5.f))
reproject(
    source=x5, destination=x4,
    src_transform=prof5["transform"], src_crs=prof5["crs"],
    dst_transform=prof4["transform"], dst_crs=prof4["crs"],
    resampling=Resampling.bilinear
)

# sr4 は sr2 の配列から作っただけで、transformはsr2由来なので prof4に合わせて再投影
sr4_on4 = to_same_grid(sr4, prof2, prof4)

mask = np.isfinite(x4) & np.isfinite(sr4_on4)
d = (sr4_on4[mask] - x4[mask]).astype(np.float64)

mae = np.mean(np.abs(d))
rmse = np.sqrt(np.mean(d**2))
bias = np.mean(d)

# 相関
x = x4[mask].astype(np.float64)
y = sr4_on4[mask].astype(np.float64)
corr = np.corrcoef(x, y)[0,1]

print("MAE[m]:", mae)
print("RMSE[m]:", rmse)
print("Bias[m]:", bias)
print("Corr  :", corr)
